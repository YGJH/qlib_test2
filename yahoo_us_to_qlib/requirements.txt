import os
import fire
from pathlib import Path
from qlib import init
from data_collector.yahoo.collector import YahooCollectorUS1d, YahooNormalize1d

# Define the directories for saving data
BASE_DIR = Path.home() / ".qlib" / "qlib_data"
SOURCE_DIR = BASE_DIR / "us_data_source"
NORMALIZE_DIR = BASE_DIR / "us_data_normalized"

# Ensure directories exist
SOURCE_DIR.mkdir(parents=True, exist_ok=True)
NORMALIZE_DIR.mkdir(parents=True, exist_ok=True)

class StockDataFetcher:
    def __init__(self, region='US', interval='1d'):
        self.region = region
        self.interval = interval

    def fetch_and_convert(self, start_date='2020-01-01', end_date=None):
        # Initialize Qlib
        init(provider_uri=str(BASE_DIR / f"{self.region.lower()}_data"))

        # Create a collector instance
        collector = YahooCollectorUS1d(
            save_dir=str(SOURCE_DIR),
            start=start_date,
            end=end_date,
            interval=self.interval,
            max_workers=4,
            delay=0.5
        )

        # Collect data
        collector.collector_data()

        # Normalize the collected data
        self.normalize_data()

    def normalize_data(self):

        normalizer = YahooNormalize1d(date_field_name='date', symbol_field_name='symbol')
